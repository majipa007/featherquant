> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `featherquant --format q4_k_m` produces a Q4_K_M GGUF byte-identical to `llama-quantize`, still inside the memory budget.

**Architecture:** K-quant math (Q4_K, Q6_K) is NOT reimplemented. A thin ctypes wrapper calls `ggml_quantize_chunk()` from the already-built `libggml-base.so` — byte-exact by construction, SIMD-fast, and it operates on exactly the row chunk we hand it, so the memory contract is unchanged. The Q4_K_M per-tensor type map is extracted empirically from a reference `llama-quantize` run and encoded as a rule function. The engine grows a small format registry.

**Tech Stack:** ctypes (stdlib), existing `gguf` package (`GGML_QUANT_SIZES` for block/type sizes), llama.cpp shared libs at `/home/sukuna/llama.cpp/build-cpu/bin/`.

## Global Constraints

- All MVP-plan global constraints still apply (Linux, CPU, single worker, budget contract, determinism).
- New runtime requirement for K-quants only: `libggml-base.so` (env `GGML_LIB` or CLI `--ggml-lib`; default `/home/sukuna/llama.cpp/build-cpu/bin/libggml-base.so`). `q8_0` format keeps working without it.
- K-quant row constraint: `ne0 % 256 == 0` (QK_K super-block). Tensors failing it keep the fallback type the reference run shows.

```python
    assert len(out) == x.size // blk * tsz
    assert out == lib.quantize_rows(x, GGMLQuantizationType.Q4_K, 256)

    out = lib.quantize_rows(x, GGMLQuantizationType.Q6_K, 256)
    blk, tsz = GGML_QUANT_SIZES[GGMLQuantizationType.Q6_K]
    assert len(out) == x.size // blk * tsz


def test_chunked_equals_full(lib):
    # rows are independent in ggml_quantize_chunk: chunking must not change bytes
    rng = np.random.default_rng(1)
    x = rng.standard_normal(8 * 256).astype(np.float32)
    full = lib.quantize_rows(x, GGMLQuantizationType.Q4_K, 256)
    parts = b"".join(lib.quantize_rows(x[i * 256:(i + 3) * 256][:3 * 256], GGMLQuantizationType.Q4_K, 256)
                     for i in (0, 3,))
    parts += lib.quantize_rows(x[6 * 256:], GGMLQuantizationType.Q4_K, 256)
    assert parts == full


def test_rejects_bad_input(lib):
    with pytest.raises(ValueError):
        lib.quantize_rows(np.zeros(100, np.float32), GGMLQuantizationType.Q4_K, 256)
```

```python
"""ctypes bridge to llama.cpp's ggml quantization kernels.

K-quant math (iterative scale search, 6-bit scale packing) is subtle and
float-summation-order-sensitive; reimplementing it byte-exact in numpy is
high-risk. Instead we call ggml_quantize_chunk() from the shared library
llama.cpp already built — byte parity by construction. The call operates on
the caller's row chunk only, so the memory contract is unchanged.
"""
import ctypes
import functools
import os

import numpy as np
from gguf import GGML_QUANT_SIZES, GGMLQuantizationType

_DEFAULT_LIB = "/home/sukuna/llama.cpp/build-cpu/bin/libggml-base.so"


class GgmlLib:
    """Loaded libggml with the one entry point we need."""

    def __init__(self, path: str):
        try:
            self._lib = ctypes.CDLL(path)
        except OSError as exc:
            raise RuntimeError(
                f"cannot load ggml library {path!r}: {exc}. Build llama.cpp "
                "or point --ggml-lib / $GGML_LIB at libggml-base.so"
            ) from exc
        fn = self._lib.ggml_quantize_chunk
        fn.restype = ctypes.c_size_t
        fn.argtypes = [ctypes.c_int, ctypes.POINTER(ctypes.c_float), ctypes.c_void_p,
                       ctypes.c_int64, ctypes.c_int64, ctypes.c_int64,
                       ctypes.POINTER(ctypes.c_float)]
        self._quantize_chunk = fn

    def quantize_rows(self, x: np.ndarray, ggml_type: GGMLQuantizationType,
                      n_per_row: int) -> bytes:
        """Quantize 1-D float32 rows to packed ggml bytes (imatrix-free)."""
        blk, tsz = GGML_QUANT_SIZES[ggml_type]
        if (x.dtype != np.float32 or x.ndim != 1 or x.size % n_per_row != 0
                or n_per_row % blk != 0):
            raise ValueError(
                f"need 1-D float32, size%{n_per_row}==0, n_per_row%{blk}==0; "
                f"got dtype={x.dtype} size={x.size}")
        nrows = x.size // n_per_row
        out = ctypes.create_string_buffer(nrows * (n_per_row // blk) * tsz)
        x = np.ascontiguousarray(x)
        written = self._quantize_chunk(
            int(ggml_type), x.ctypes.data_as(ctypes.POINTER(ctypes.c_float)),
            out, 0, nrows, n_per_row, None)
        if written != len(out):
            raise RuntimeError(f"ggml_quantize_chunk wrote {written}, expected {len(out)}")
        return out.raw


@functools.lru_cache(maxsize=4)
def load_ggml(path: str | None = None) -> GgmlLib:
    """Load (and cache) the ggml library from path, $GGML_LIB, or the default."""
    return GgmlLib(path or os.environ.get("GGML_LIB", _DEFAULT_LIB))
```

```python
"""Output format registry: model-level preset -> per-tensor ggml type."""
from dataclasses import dataclass
from typing import Callable

from gguf import GGMLQuantizationType, LlamaFileType

from .gguf_io import ITEMSIZE
from .q8_0 import BLOCK

QK_K = 256  # K-quant super-block length


def _quantizable(t, blk: int) -> bool:
    """2-D+ float tensor whose contiguous row divides the block length."""
    return (len(t.shape) >= 2 and t.tensor_type in ITEMSIZE
            and int(t.shape[0]) % blk == 0)


def _q8_0_rule(t, n_layers: int) -> GGMLQuantizationType:
    return GGMLQuantizationType.Q8_0 if _quantizable(t, BLOCK) else t.tensor_type


def _q4_k_m_rule(t, n_layers: int) -> GGMLQuantizationType:
    # Encoded from tests/data/qwen3-0.6b-q4_k_m-types.json (llama-quantize
    # ground truth), NOT from llama.cpp source. Expected shape of the rules:
    #   not quantizable            -> source type
    #   token_embd / output.weight -> Q6_K
    #   attn_v.weight              -> Q6_K
    #   ffn_down.weight, layer in first/last band or every 3rd -> Q6_K
    #   everything else quantizable -> Q4_K
    # ADJUST to make test_q4_k_m_matches_reference_dump pass; the dump wins.
    if not _quantizable(t, QK_K):
        return _q8_0_rule(t, n_layers)  # narrow tensors: dump shows fallback
    ...


@dataclass(frozen=True)
class FormatSpec:
    file_type: LlamaFileType
    tensor_type: Callable
    needs_ggml: bool


FORMATS: dict[str, FormatSpec] = {
    "q8_0": FormatSpec(LlamaFileType.MOSTLY_Q8_0, _q8_0_rule, needs_ggml=False),
    "q4_k_m": FormatSpec(LlamaFileType.MOSTLY_Q4_K_M, _q4_k_m_rule, needs_ggml=True),
}
```
